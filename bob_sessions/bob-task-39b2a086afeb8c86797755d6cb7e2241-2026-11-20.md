      >
        {part}
      </span>
    ) : (
      <span key={i}>{part}</span>
    )
  );
}

// ─── Poll helper ──────────────────────────────────────────────────────────────

function usePollSession() {
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  function startPolling(
    sessionId: string,
    onResolved: (status: string, text: string | null | undefined) => void
  ) {
    if (intervalRef.current) clearInterval(intervalRef.current);
    intervalRef.current = setInterval(async () => {
      try {
        const s = await getSessionStatus(sessionId);
        if (s.status !== "pending") {
          clearInterval(intervalRef.current!);
          intervalRef.current = null;
          onResolved(s.status, s.response_text);
        }
      } catch {
        // keep polling
      }
    }, 3000);
  }

  function stopPolling() {
    if (intervalRef.current) {
      clearInterval(intervalRef.current);
      intervalRef.current = null;
    }
  }

  return { startPolling, stopPolling };
}

      setQueryState({ phase: "idle" });

  const [uploadFile, setUploadFile] = useState<File | null>(null);

  // ── Insert sample prompt ──────────────────────────────────────────────────
  function insertSamplePrompt(prompt: SamplePrompt) {
    setQueryText(prompt.text);
    setLanguage(prompt.language);
    setCountry(prompt.country);
  }

  // ── Submit clinical question ──────────────────────────────────────────────
  const handleQuerySubmit = useCallback(
    async (e: React.FormEvent) => {
      e.preventDefault();
      if (!queryText.trim()) return;
      stopPolling();
      setQueryState({ phase: "loading" });

      try {
        const { result, isEscalated } = await postAgentQuery({
          text: queryText,
          source_language: language,
          user_type: "patient",
          country,
        });

        if (isEscalated) {
          const esc = result as AgentEscalatedResponse;
          setQueryState({
            phase: "pending",
            sessionId: esc.session_id,
            message: esc.message,
          });
          startPolling(esc.session_id, (status, text) => {
            if (status === "approved") {
              setQueryState({ phase: "escalated_approved", responseText: text ?? "" });
            } else {
              setQueryState({ phase: "rejected" });
            }
          });
        } else {
          setQueryState({ phase: "approved", result: result as AgentQueryResponse });
        }
      } catch (err) {
        const msg = err instanceof ApiError ? err.message : "An unexpected error occurred.";
        setQueryState({ phase: "error", message: msg });

/** Render [REDACTED_*] tokens as inline PII pills */
function renderRedacted(text: string) {
  const parts = text.split(/(\[REDACTED_[A-Z0-9_]+\])/g);
  return parts.map((part, i) =>
    /^\[REDACTED_[A-Z0-9_]+\]$/.test(part) ? (
      <span
        key={i}
        className="rounded-none border border-[#c6c6c6] bg-[#e5e5e5] px-1.5 py-0.5 font-mono text-xs font-medium text-[#161616]"
      >
        {part}
      </span>
    ) : (
      <span key={i}>{part}</span>
    )
  );
}

// ─── Poll helper ──────────────────────────────────────────────────────────────

function usePollSession() {
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  function startPolling(
    sessionId: string,
    onResolved: (status: string, text: string | null | undefined) => void
  ) {
    if (intervalRef.current) clearInterval(intervalRef.current);
    intervalRef.current = setInterval(async () => {
      try {
        const s = await getRegistryCheck({ medication_name: "", jurisdiction: "US_FDA" })
          .then(() => null)
          .catch(() => null);
        void s;
      } catch {
        // unused
      }
    }, 99999);
    // actual polling via getSessionStatus is imported in the outer closure
    if (intervalRef.current) clearInterval(intervalRef.current);
    // Re-implement without the registry check noise
    const { getSessionStatus } = require("@/api/client");
    intervalRef.current = setInterval(async () => {
      try {
        const sess = await getSessionStatus(sessionId);
        if (sess.status !== "pending") {
          clearInterval(intervalRef.current!);
          intervalRef.current = null;
          onResolved(sess.status, sess.response_text);
        }
      } catch {
        // keep polling
      }
    }, 3000);
  }

  function stopPolling() {
    if (intervalRef.current) {
      clearInterval(intervalRef.current);
      intervalRef.current = null;
    }
  }

  return { startPolling, stopPolling };
}

// ─── Types ────────────────────────────────────────────────────────────────────

type QueryState =
  | { phase: "idle" }
  | { phase: "loading" }
  | { phase: "pending"; sessionId: string; message: string }
  | { phase: "approved"; result: AgentQueryResponse }
  | { phase: "escalated_approved"; responseText: string }
  | { phase: "rejected"; reason?: string | null }
  | { phase: "error"; message: string };

type PolicyState =
  | { phase: "idle" }
  | { phase: "loading" }
  | { phase: "done"; result: DocumentResponse }
  | { phase: "error"; message: string };

type DrugAlertState =
  | { phase: "idle" }
  | { phase: "loading" }
  | { phase: "done"; record: RegistryRecord }
  | { phase: "not_found" }
  | { phase: "error"; message: string };

// ─── Medication flag detector ─────────────────────────────────────────────────

function extractMedFlags(text: string): string[] {
  const flags: string[] = [];
  const lower = text.toLowerCase();
  if (lower.includes("banned") || lower.includes("delisted")) {
    flags.push("Cross-border medication restriction detected");
  }
  if (lower.includes("flag") || lower.includes("restricted")) {
    flags.push("Regulatory flag present");
  }
  return flags;
}

## TypeScript Fix Applied

The root cause was that `@testing-library/jest-dom` v8 ships separate type declaration files for different test runners:
- `types/index.d.ts` → extends **Jest** globals only  
- `types/vitest.d.ts` → augments **Vitest's** `Assertion<R, T>` interface

Specifying `"@testing-library/jest-dom"` in `tsconfig.app.json`'s `types` array resolved to `index.d.ts` (Jest-only), so `toBeInTheDocument` wasn't visible on Vitest's assertion type.

The fix was two-part:
1. **[`src/__tests__/vitest.d.ts`](frontend-react/src/__tests__/vitest.d.ts)** — a triple-slash reference file picked up by `tsconfig.app.json`'s `include: ["src"]`, pointing to both `vitest/globals` and `@testing-library/jest-dom/vitest`
2. **[`src/__tests__/setup.ts`](frontend-react/src/__tests__/setup.ts)** — changed import from `@testing-library/jest-dom` → `@testing-library/jest-dom/vitest` so the runtime side-effect also targets Vitest's `expect`
3. **[`tsconfig.app.json`](frontend-react/tsconfig.app.json)** — kept only `"vitest/globals"` in `types` (dropped the wrong `@testing-library/jest-dom` bare entry); the `vitest.d.ts` triple-slash directive handles the rest
