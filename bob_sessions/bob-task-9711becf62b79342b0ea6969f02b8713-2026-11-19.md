backend/watsonx/client.py
──────────────────────────
Thin wrapper around ibm-watsonx-ai for Granite model inference.

Design principles
─────────────────
* Real mode (WATSONX_MOCK_MODE=false):  calls ibm-watsonx-ai ModelInference,
  raises WatsonxClientError on credential/network failures.
* Mock mode (WATSONX_MOCK_MODE=true):  returns deterministic stub responses
  so tests and offline demos never burn cloud credits.
* Every call returns a WatsonxResponse dataclass that includes the response
  text, an estimated confidence score, and the raw model metadata.

Usage
─────
    from backend.watsonx.client import get_client
    client = get_client()
    resp = client.generate(prompt="Explain deductibles in plain English.")
    print(resp.text, resp.confidence)
"""

from __future__ import annotations

import logging
import re
from dataclasses import dataclass, field
from functools import lru_cache
from typing import Any, Dict, Optional

from backend.config import get_settings

logger = logging.getLogger(__name__)


# ──────────────────────────────────────────────────────────────────────────────
# Public error type
# ──────────────────────────────────────────────────────────────────────────────

class WatsonxClientError(RuntimeError):
    """Raised when a watsonx.ai API call fails."""


# ──────────────────────────────────────────────────────────────────────────────
# Response dataclass
# ──────────────────────────────────────────────────────────────────────────────

@dataclass
class WatsonxResponse:
    """
    Normalised response from a watsonx.ai generate call.

    Attributes
    ──────────
    text        : Generated text (stripped).
    confidence  : Estimated confidence score 0.0–1.0.
                  Derived from the token log-probabilities when available;
                  defaults to 0.90 for successful real calls and a fixed
                  value in mock mode.
    model_id    : The model identifier that produced the response.
    mock        : True when this response came from the stub fallback.
    raw         : The original SDK response dict (None in mock mode).
    """

    text: str
    confidence: float
    model_id: str
    mock: bool = False
    raw: Optional[Dict[str, Any]] = field(default=None, repr=False)


# ──────────────────────────────────────────────────────────────────────────────
# Default generation parameters for Granite
# ──────────────────────────────────────────────────────────────────────────────

_DEFAULT_PARAMS: Dict[str, Any] = {
    "max_new_tokens": 512,
    "temperature": 0.3,
    "top_p": 0.9,
    "repetition_penalty": 1.1,
}

# Client implementation
# ──────────────────────────────────────────────────────────────────────────────

class WatsonxClient:
    """
    Wraps ibm-watsonx-ai ModelInference with mock-mode fallback.

    Instantiate via ``get_client()`` to reuse the singleton across the
    request lifecycle.
    """

    def __init__(self) -> None:
        self._settings = get_settings()
        self._mock = self._settings.watsonx_mock_mode
        self._model_id = self._settings.watsonx_model_id
        self._model: Any = None  # lazy-initialised in _ensure_model()

        if not self._mock:
            self._validate_credentials()

    # ── Credential guard ──────────────────────────────────────────────────────

    def _validate_credentials(self) -> None:
        """Raise early if required credentials are absent in real mode."""
        missing = [
            name
            for name, attr in [
                ("WATSONX_API_KEY", self._settings.watsonx_api_key),
                ("WATSONX_PROJECT_ID", self._settings.watsonx_project_id),
            ]
            if not attr
        ]
        if missing:
            raise WatsonxClientError(
                f"Real mode requires: {', '.join(missing)}. "
                "Set them in .env or set WATSONX_MOCK_MODE=true for offline use."
            )

    # ── Lazy SDK initialisation ───────────────────────────────────────────────

    def _ensure_model(self) -> Any:
        """Return the ModelInference instance, creating it on first use."""
        if self._model is None:
            try:
                from ibm_watsonx_ai import Credentials
                from ibm_watsonx_ai.foundation_models import ModelInference
            except ImportError as exc:
                raise WatsonxClientError(
                    "ibm-watsonx-ai package is not installed. "
                    "Run: pip install ibm-watsonx-ai"
                ) from exc

            credentials = Credentials(
                api_key=self._settings.watsonx_api_key,
                url=self._settings.watsonx_url,
            )
            self._model = ModelInference(
                model_id=self._model_id,
                credentials=credentials,
                project_id=self._settings.watsonx_project_id,
                params=_DEFAULT_PARAMS,
            )
            logger.info(
                "watsonx ModelInference initialised: model=%s url=%s",
                self._model_id,
                self._settings.watsonx_url,
            )
        return self._model

    # ── Confidence estimation ─────────────────────────────────────────────────

    @staticmethod
    def _estimate_confidence(raw_response: Dict[str, Any]) -> float:
        """
        Derive a confidence estimate from the SDK response.

        Uses the mean of token log-probabilities when present; falls back to
        0.90 for a successful call with no log-prob metadata.
        """
        try:
            results = raw_response.get("results", [])
            if not results:
                return 0.90
            tokens = results[0].get("generated_tokens", [])
            if not tokens:
                return 0.90
            log_probs = [
                t.get("logprob", 0.0)
                for t in tokens
                if t.get("logprob") is not None
            ]
            if not log_probs:
                return 0.90
            import math
            avg_lp = sum(log_probs) / len(log_probs)
            # exp(avg_log_prob) gives an approximate per-token probability
            confidence = min(1.0, max(0.0, math.exp(avg_lp)))
            return round(confidence, 4)
        except Exception:
            return 0.90

    # ── Mock stubs ────────────────────────────────────────────────────────────

    def _mock_generate(self, prompt: str) -> WatsonxResponse:
        """
        Return a deterministic stub response without touching the network.

        The stub echoes a summary of the first 80 chars of the prompt so
        tests can assert on content without hitting watsonx.ai.
        """
        truncated = prompt[:80].replace("\n", " ")
        stub_text = (
            f"[MOCK] Processed: \"{truncated}\" — "
            "clinical context understood, proceeding with guidance."
        )
        return WatsonxResponse(
            text=stub_text,
            confidence=0.95,
            model_id=self._model_id,
            mock=True,
            raw=None,
        )

    # ── Public API ────────────────────────────────────────────────────────────

    def generate(
        self,
        prompt: str,
        *,
        max_new_tokens: Optional[int] = None,
        temperature: Optional[float] = None,
    ) -> WatsonxResponse:
        """
        Generate text from *prompt* using Granite.

        Parameters
        ──────────
        prompt         : The full prompt string.
        max_new_tokens : Override the default max_new_tokens for this call.
        temperature    : Override the default temperature for this call.

        Returns
        ───────
        WatsonxResponse with generated text and confidence score.

        Raises
        ──────
        WatsonxClientError — on API or credential failure (real mode only).
        """
        if self._mock:
            logger.debug("Mock mode: skipping real API call.")
            return self._mock_generate(prompt)

        try:
            model = self._ensure_model()
            params: Dict[str, Any] = {}
            if max_new_tokens is not None:
                params["max_new_tokens"] = max_new_tokens
            if temperature is not None:
                params["temperature"] = temperature

            raw = model.generate(prompt=prompt, params=params or None)
            results = raw.get("results", [{}])
            generated_text = results[0].get("generated_text", "").strip()
            confidence = self._estimate_confidence(raw)

            logger.debug(
                "watsonx generate: model=%s tokens=%d confidence=%.4f",
                self._model_id,
                results[0].get("generated_token_count", -1),
                confidence,
            )
            return WatsonxResponse(
                text=generated_text,
                confidence=confidence,
                model_id=self._model_id,
                mock=False,
                raw=raw,
            )
        except WatsonxClientError:
            raise
        except Exception as exc:
            raise WatsonxClientError(
                f"watsonx.ai API call failed: {exc}"
            ) from exc

    @property
    def model_id(self) -> str:
        return self._model_id

    @property
    def is_mock(self) -> bool:
        return self._mock


# ──────────────────────────────────────────────────────────────────────────────
# Singleton accessor
# ──────────────────────────────────────────────────────────────────────────────

@lru_cache(maxsize=1)
def get_client() -> WatsonxClient:
    """
    Return the application-wide WatsonxClient singleton.

    Call ``get_client.cache_clear()`` in tests to reset between cases.
    """
    return WatsonxClient()

def _redact_names(text: str) -> Tuple[str, int]:
    """Apply name heuristic, skipping known non-name phrases."""
    count = 0

    def replace_name(match: re.Match) -> str:  # type: ignore[type-arg]
        nonlocal count
        matched = match.group(0)
        # Skip if the matched string is in the exclusion list
        if matched in _NAME_EXCLUSIONS:
            return matched
        count += 1
        return "[NAME]"

    result = _NAME_HEURISTIC.sub(replace_name, text)
    return result, count

def redact(text: str) -> RedactionResult:
    """
    Strip PII from *text* and return a RedactionResult.

    The original *text* is never stored or returned — only the redacted
    version and summary statistics are exposed.

    Parameters
    ──────────
    text : Raw user input (may contain PII).

    Returns
    ───────
    RedactionResult with redacted_text, redaction_count, types_found, and
    has_residual_pii flag.
    """
    if not text:
        return RedactionResult(
            redacted_text="",
            redaction_count=0,
            types_found=[],
            has_residual_pii=False,
        )

    working = text
    total_count = 0
    types: List[str] = []

    # Apply all structured patterns in order
    for pattern, placeholder, pii_type in _PATTERNS:
        new_text, n = _apply_pattern(working, pattern, placeholder)
        if n:
            total_count += n
            types.append(pii_type)
            working = new_text

    # Apply name heuristic last (after structured PII is already masked)
    working, name_count = _redact_names(working)
    if name_count:
        total_count += name_count
        types.append("name")

    # Residual-PII heuristic: check for standalone dates that may be DOBs
    # We flag but do NOT redact (too aggressive — could be appointment dates)
    residual = bool(_DATE_STANDALONE.search(working))

    return RedactionResult(
        redacted_text=working,
        redaction_count=total_count,
        types_found=types,
        has_residual_pii=residual,
    )


def _apply_pattern(
    text: str, pattern: re.Pattern, placeholder: str  # type: ignore[type-arg]
) -> Tuple[str, int]:
    """Replace all matches of *pattern* with *placeholder*; return (new_text, count)."""
    count = 0

    def _replace(_match: re.Match) -> str:  # type: ignore[type-arg]
        nonlocal count
        count += 1
        return placeholder

    new_text = pattern.sub(_replace, text)
    return new_text, count

    """

    Granite in the full pipeline; this provides a fast pre-check.
    """
    tokens = set(re.findall(r"\b\w+\b", text.lower()))
    scores = {
        lang: sum(1 for w in markers if w in tokens)
        for lang, markers in _LANG_MARKERS.items()

def evaluate_escalation(inp: EscalationInput) -> EscalationResult:
    """
    Evaluate all five HITL escalation rules against *inp*.

    Parameters
    ──────────
    inp : EscalationInput with all pipeline stage outputs.

    Returns
    ───────
    EscalationResult.  Call ``enqueue_if_escalated()`` to actually push
    to the HITL queue.
    """
    settings = get_settings()
    threshold = settings.hitl_confidence_threshold
    rules_triggered: List[str] = []

    # Rule 1 — Low confidence
    if inp.min_confidence < threshold:
        rules_triggered.append(
            f"rule_1_low_confidence (score={inp.min_confidence:.4f} < {threshold})"
        )

    # Rule 2 — Banned or restricted medication
    for flag in inp.medication_flags:
        if flag.status in ("banned", "restricted"):
            rules_triggered.append(
                f"rule_2_banned_medication ({flag.name} is {flag.status} "
                f"in {flag.jurisdiction})"
            )
            break  # one trigger is enough per category

    # Rule 3 — Residual PII
    if inp.redaction_result.has_residual_pii:
        rules_triggered.append("rule_3_residual_pii")

    # Rule 4 — High-stakes clinical topic
    if _has_high_stakes_topic(inp.original_text):
        rules_triggered.append("rule_4_high_stakes_topic")

    # Rule 5 — Ambiguous jurisdiction
    if inp.is_ambiguous_jurisdiction:
        rules_triggered.append(
            f"rule_5_ambiguous_jurisdiction (country={inp.country!r})"
        )

    should_escalate = bool(rules_triggered)

    if should_escalate:
        logger.info(
            "HITL escalation triggered for session=%s rules=%s",
            inp.session_id,
            rules_triggered,
        )
    else:
        logger.debug("No HITL escalation for session=%s", inp.session_id)

    return EscalationResult(
        should_escalate=should_escalate,
        rules_triggered=rules_triggered,
        escalation_input=inp if should_escalate else None,
    )


def enqueue_if_escalated(result: EscalationResult) -> Optional[str]:
    """
    If *result* requires escalation, push it to the HITL queue and return
    the review_id.  Returns None when no escalation is needed.

    Import is deferred to avoid a circular dependency between pipeline and hitl.
    """
    if not result.should_escalate or result.escalation_input is None:
        return None

    from backend.hitl.queue import get_queue  # deferred import

    inp = result.escalation_input
    review_id = get_queue().enqueue(
        session_id=inp.session_id,
        user_type=inp.user_type,
        country=inp.country,
        pipeline_trace=inp.pipeline_trace,
        draft_response=inp.draft_response,
        escalation_rules=result.rules_triggered,
    )
    return review_id

backend/hitl/queue.py
──────────────────────
Thread-safe in-memory HITL review queue.

Architecture note (architecture.md §12):
  "In-memory list (MVP) — Simplest viable; documented upgrade path to Redis
  for production."

The queue is a module-level singleton protected by a threading.Lock so it is

Usage

    from backend.hitl.queue import get_queue

    q = get_queue()
    review_id = q.enqueue(session_id=..., user_type=..., ...)
    item = q.get_pending(review_id)
    q.approve(review_id, approved_response="...", reviewer_note=None)
    q.reject(review_id, rejection_reason="...")
"""

from __future__ import annotations

import threading
import uuid
from datetime import datetime, timezone
from typing import Any, Dict, List, Optional

from backend.hitl.models import ReviewItem, ReviewStatus


class HITLQueue:
    """
    Thread-safe in-memory HITL review queue.

    All public methods acquire the internal lock so they are safe to call
    from concurrent FastAPI request handlers.
    """

    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._items: Dict[str, ReviewItem] = {}  # review_id -> ReviewItem

    # ── Write operations ──────────────────────────────────────────────────────

    def enqueue(
        self,
        session_id: str,
        user_type: str,
        country: str,
        pipeline_trace: Dict[str, Any],
        draft_response: str,
        escalation_rules: Optional[List[str]] = None,
    ) -> str:
        """
        Add a new item to the queue.

        Parameters
        ──────────
        session_id       : Session UUID from the agent request.
        user_type        : "patient" | "provider" | "coordinator".
        country          : ISO 3166-1 alpha-2 country code.
        pipeline_trace   : Stage outputs dict (no raw PII).
        draft_response   : Draft response text for the reviewer.
        escalation_rules : List of triggered rule labels.

        Returns
        ───────
        review_id (UUID string) for the new queue item.
        """
        review_id = str(uuid.uuid4())
        item = ReviewItem(
            review_id=review_id,
            session_id=session_id,
            submitted_at=datetime.now(tz=timezone.utc),
            user_type=user_type,
            country=country,
            pipeline_trace=pipeline_trace,
            draft_response=draft_response,
            escalation_rules=escalation_rules or [],
            status=ReviewStatus.pending,
        )
        with self._lock:
            self._items[review_id] = item
        return review_id

    def approve(
        self,
        review_id: str,
        approved_response: str,
        reviewer_note: Optional[str] = None,
    ) -> ReviewItem:
        """
        Mark a pending item as approved with the reviewer's final response.

        Raises
        ──────
        KeyError  — when *review_id* does not exist.
        ValueError — when item is not in 'pending' status.
        """
        with self._lock:
            item = self._items.get(review_id)
            if item is None:
                raise KeyError(f"review_id not found: {review_id!r}")
            if item.status != ReviewStatus.pending:
                raise ValueError(
                    f"Cannot approve item with status={item.status!r}. "
                    "Only 'pending' items can be approved."
                )
            updated = item.model_copy(update={
                "status": ReviewStatus.approved,
                "approved_response": approved_response,
                "reviewer_note": reviewer_note,
                "reviewed_at": datetime.now(tz=timezone.utc),
            })
            self._items[review_id] = updated
        return updated

    def reject(self, review_id: str, rejection_reason: str) -> ReviewItem:
        """
        Mark a pending item as rejected.

        Raises
        ──────
        KeyError  — when *review_id* does not exist.
        ValueError — when item is not in 'pending' status.
        """
        with self._lock:
            item = self._items.get(review_id)
            if item is None:
                raise KeyError(f"review_id not found: {review_id!r}")
            if item.status != ReviewStatus.pending:
                raise ValueError(
                    f"Cannot reject item with status={item.status!r}. "
                    "Only 'pending' items can be rejected."
                )
            updated = item.model_copy(update={
                "status": ReviewStatus.rejected,
                "rejection_reason": rejection_reason,
                "reviewed_at": datetime.now(tz=timezone.utc),
            })
            self._items[review_id] = updated
        return updated

    # ── Read operations ───────────────────────────────────────────────────────

    def get_pending(self, review_id: Optional[str] = None) -> List[ReviewItem]:
        """
        Return pending items.

        When *review_id* is provided, return a list containing that specific
        item (empty list if not found or not pending).
        When *review_id* is None, return all pending items ordered by
        submitted_at ascending.
        """
        with self._lock:
            if review_id is not None:
                item = self._items.get(review_id)
                if item and item.status == ReviewStatus.pending:
                    return [item]
                return []
            return sorted(
                (i for i in self._items.values() if i.status == ReviewStatus.pending),
                key=lambda i: i.submitted_at,
            )

    def get_item(self, review_id: str) -> Optional[ReviewItem]:
        """Return any item by review_id regardless of status."""
        with self._lock:
            return self._items.get(review_id)

    def get_by_session(self, session_id: str) -> Optional[ReviewItem]:
        """Return the most recent review item for a given session_id."""
        with self._lock:
            matches = [
                i for i in self._items.values() if i.session_id == session_id
            ]
        if not matches:
            return None
        return max(matches, key=lambda i: i.submitted_at)

    def pending_count(self) -> int:
        """Return the number of items currently in 'pending' status."""
        with self._lock:
            return sum(
                1 for i in self._items.values() if i.status == ReviewStatus.pending
            )

    def clear(self) -> None:
        """Remove all items. Intended for testing only."""
        with self._lock:
            self._items.clear()


# ──────────────────────────────────────────────────────────────────────────────
# Module-level singleton
# ──────────────────────────────────────────────────────────────────────────────

_queue: Optional[HITLQueue] = None
_queue_lock = threading.Lock()


def get_queue() -> HITLQueue:
    """
    Return the application-wide HITLQueue singleton.

    Thread-safe double-checked initialisation.
    """
    global _queue
    if _queue is None:
        with _queue_lock:
            if _queue is None:
                _queue = HITLQueue()
    return _queue

def agent_query(body: AgentQueryRequest, response: Response):
    """
    Main entry point for the full agentic pipeline.

    Returns **200 OK** with the full pipeline trace when no escalation is
    needed, or **202 Accepted** with a review_id when HITL review is required.
    """
    session_id = body.session_id or str(uuid.uuid4())
    client = get_client()

    # ── Stage 0: PII Redaction ────────────────────────────────────────────────
    redaction = redact(body.text)
    redacted_text = redaction.redacted_text

    # ── Jurisdiction resolution (deterministic, no LLM) ──────────────────────
    jurisdiction = resolve(body.country)

    # ── Stage 1: Translation (redacted text -> English for downstream stages) ─
    translation_result = translate(
        text=redacted_text,
        source_language=body.source_language,
        target_language="en",
        client=client,
    )
    english_text = translation_result.translated_text

    # ── Stage 2: Medical Interpretation ──────────────────────────────────────
    interpretation = interpret(
        text=english_text,
        jurisdiction=jurisdiction,
        client=client,
    )

    # ── Stage 3: Cultural Context ─────────────────────────────────────────────
    cultural = add_cultural_context(
        text=interpretation.plain_language_text,
        jurisdiction=jurisdiction,
        user_type=body.user_type,
        client=client,
    )

    # ── Stage 4: Next Steps ────────────────────────────────────────────────────
    next_steps = generate_next_steps(
        framed_text=cultural.framed_text,
        user_type=body.user_type,
        medication_flags=interpretation.medication_flags,
        jurisdiction=jurisdiction,
        client=client,
    )

    # Assemble draft response (next steps in English, to be translated below)
    draft_english = next_steps.next_steps_text

    # Translate back to source language if needed
    if body.source_language.lower() != "en":
        back_translate = translate(
            text=draft_english,
            source_language="en",
            target_language=body.source_language,
            client=client,
        )
        final_response_text = back_translate.translated_text
        min_confidence = min(
            translation_result.confidence,
            interpretation.confidence,
            cultural.confidence,
            next_steps.confidence,
            back_translate.confidence,
        )
    else:
        final_response_text = draft_english
        min_confidence = min(
            translation_result.confidence,
            interpretation.confidence,
            cultural.confidence,
            next_steps.confidence,
        )

    # Build pipeline trace (never includes raw PII)
    pipeline_trace_dict: Dict[str, Any] = {
        "jurisdiction": jurisdiction.registry,
        "pii_redacted": redaction.redaction_count > 0,
        "translation": translation_result.translated_text,
        "medical_interpretation": interpretation.plain_language_text,
        "cultural_context": cultural.framed_text,
        "next_steps": next_steps.next_steps_text,
    }

    escalation_input = EscalationInput(
        session_id=session_id,
        original_text=redacted_text,     # already redacted
        user_type=body.user_type,
        country=body.country,
        is_ambiguous_jurisdiction=jurisdiction.is_ambiguous,
        redaction_result=redaction,
        min_confidence=min_confidence,
        medication_flags=interpretation.medication_flags,
        pipeline_trace=pipeline_trace_dict,
        draft_response=final_response_text,
    )

    review_id = enqueue_if_escalated(escalation_result)

            session_id=session_id,
            escalated=True,
            review_id=review_id,
            status="pending_review",
            message=_escalation_message(body.source_language),
        )

    return AgentQueryResponse(
        session_id=session_id,
        pipeline_trace=PipelineTrace(**pipeline_trace_dict),
        escalated=False,
        review_id=None,
        response_text=final_response_text,
    )

class TestHITLQueueThreadSafety:

    def test_concurrent_enqueues_no_data_loss(self, queue):
        """50 concurrent enqueues must all succeed without race conditions."""
        n = 50
        results: list = []
        errors: list = []

        def _enqueue():
            try:
                rid = queue.enqueue(
                    session_id=str(uuid.uuid4()),
                    user_type="patient",
                    country="US",
                    pipeline_trace={},
                    draft_response="draft",
                )
                results.append(rid)
            except Exception as exc:
                errors.append(exc)

        threads = [threading.Thread(target=_enqueue) for _ in range(n)]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        assert len(errors) == 0
        assert len(results) == n
        assert queue.pending_count() == n
        # All review_ids must be unique
        assert len(set(results)) == n
