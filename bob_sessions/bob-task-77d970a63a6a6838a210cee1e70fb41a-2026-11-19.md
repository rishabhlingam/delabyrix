    ("consent_icu_admission", "es"): "Formulario de Consentimiento de Admisión a UCI",
    ("consent_icu_admission", "hi"): "आईसीयू प्रवेश सहमति फॉर्म",

    ("consent_surgery", "hi"): "शल्य प्रक्रिया सहमति फॉर्म",

# ──────────────────────────────────────────────────────────────────────────────

class DocumentResponse(BaseModel):
    doc_type: str
    language: str
    title: str
    content: str


# ──────────────────────────────────────────────────────────────────────────────
# GET /documents/{doc_type}/{language}
# ──────────────────────────────────────────────────────────────────────────────

@router.get(
    "/{doc_type}/{language}",
    response_model=DocumentResponse,
    summary="Retrieve a synthetic consent or billing document",
)
def get_document(doc_type: str, language: str) -> DocumentResponse:
    """
    Retrieve a synthetic multilingual document by type and language.

    - **doc_type**: `consent_icu_admission` | `consent_surgery` | `billing_deductible_explanation`
    - **language**: `en` | `es` | `hi`
    """
    if doc_type not in _VALID_DOC_TYPES:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=(
                f"Invalid doc_type={doc_type!r}. "
                f"Valid values: {sorted(_VALID_DOC_TYPES)}"
            ),
        )
    if language not in _VALID_LANGUAGES:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=(
                f"Invalid language={language!r}. "
                f"Valid values: {sorted(_VALID_LANGUAGES)}"
            ),
        )

    file_path = _DOCS_ROOT / language / f"{doc_type}.txt"
    if not file_path.exists():
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Document file not found: {file_path.name}",
        )

    content = file_path.read_text(encoding="utf-8")
    title = _TITLES.get((doc_type, language), doc_type.replace("_", " ").title())

    return DocumentResponse(
        doc_type=doc_type,
        language=language,
        title=title,
        content=content,
    )

        default=None,
        description="RNG seed for reproducibility (None = use module default)",
    )
    medication_count: int = Field(
        default=50,
        description="Number of medication records (default 50; actual value is fixed by generator)",
    )

    counts: GenerateCounts

# POST /synthetic/generate
# ──────────────────────────────────────────────────────────────────────────────

    "/generate",
    response_model=GenerateResponse,
    summary="Trigger a fresh synthetic data generation run",
)

    """
    Regenerate all synthetic data files and reload the in-memory registry.

    - Overwrites `synthetic_data/medication_registry.json`
    - Overwrites `synthetic_data/patient_logs.json`
    - Overwrites all 9 consent/billing documents in `synthetic_data/consent_docs/`

    The in-memory medication registry is reloaded immediately after generation.
    """

    root.mkdir(parents=True, exist_ok=True)

    # ── 1. Medication registry ────────────────────────────────────────────────
    med_records = generate_medication_registry()
    med_path = root / "medication_registry.json"
    med_path.write_text(json.dumps(med_records, indent=2, default=str), encoding="utf-8")

    patient_logs = generate_patient_logs(body.patient_log_count)
    logs_path = root / "patient_logs.json"

    docs = generate_consent_billing_docs()
    doc_count = 0

        doc_count += 1

    # ── 4. Reload in-memory registry ─────────────────────────────────────────
    reload_registry()

    return GenerateResponse(
        status="generated",
        counts=GenerateCounts(
            medications=len(med_records),
            patient_logs=len(patient_logs),
            documents=doc_count,
        ),
    )

class TestE2ESpanishPatientBannedDrug:
    """
    Full end-to-end flow:
      1. Spanish patient submits a query mentioning metamizole (banned in US_FDA)
      2. Agent pipeline returns 202 — escalated for HITL review
      3. Item appears in GET /hitl/queue with correct metadata
      4. Care coordinator approves with an edited response
      5. Patient polling GET /agent/session/{id} returns approved response
    """

    def test_e2e_spanish_banned_drug_full_flow(self, client):
        # ── Step 1: Spanish patient query ─────────────────────────────────────
        session_id = str(uuid.uuid4())
        query_resp = _agent_query(
            client,
            text="Tomo metamizol en casa para el dolor. ¿Puedo usarlo aquí en los Estados Unidos?",
            source_language="es",
            country="US",
            user_type="patient",
            session_id=session_id,
        )

        # Must be escalated due to metamizole being banned in US_FDA (rule 2)
        assert query_resp.status_code == 202, \
            f"Expected 202 escalation, got {query_resp.status_code}: {query_resp.text}"
        escalation_data = query_resp.json()
        assert escalation_data["escalated"] is True
        assert "review_id" in escalation_data
        assert escalation_data["status"] == "pending_review"
        # Localised notice should be in Spanish
        assert "solicitud" in escalation_data["message"].lower() or \
               "revisión" in escalation_data["message"].lower() or \
               "clínico" in escalation_data["message"].lower()

        returned_session_id = escalation_data["session_id"]
        review_id = escalation_data["review_id"]

        # ── Step 2: Item appears in HITL queue ────────────────────────────────
        queue_resp = client.get("/hitl/queue")
        assert queue_resp.status_code == 200
        queue_data = queue_resp.json()
        assert queue_data["pending_count"] >= 1

        queue_items = queue_data["items"]
        matching = [i for i in queue_items if i["review_id"] == review_id]
        assert len(matching) == 1, \
            f"review_id {review_id!r} not found in queue items"

        item = matching[0]
        assert item["session_id"] == returned_session_id
        assert item["user_type"] == "patient"
        assert item["country"] == "US"
        # Pipeline trace must exist and not contain raw PII patterns
        trace = item.get("pipeline_trace", {})
        assert "jurisdiction" in trace
        trace_str = json.dumps(trace)
        # The original Spanish text must not appear verbatim
        assert "metamizol en casa" not in trace_str.lower()

        # ── Step 3: Coordinator polls — should be pending ─────────────────────
        poll_resp = client.get(f"/agent/session/{returned_session_id}")
        assert poll_resp.status_code == 200
        poll_data = poll_resp.json()
        assert poll_data["status"] == "pending"
        assert poll_data["response_text"] is None

        # ── Step 4: Coordinator approves ──────────────────────────────────────
        approved_text = (
            "Metamizol (dipirona) está prohibido en los Estados Unidos. "
            "Por favor consulte a su médico para alternativas seguras como "
            "ibuprofeno o acetaminofén."
        )
        approve_resp = client.post(
            "/hitl/approve",
            json={
                "review_id": review_id,
                "approved_response": approved_text,
                "reviewer_note": "Confirmed: metamizole banned in US_FDA. Alternatives provided.",
            },
        )
        assert approve_resp.status_code == 200
        assert approve_resp.json()["status"] == "approved"

        # ── Step 5: Patient polling returns approved response ─────────────────
        final_poll = client.get(f"/agent/session/{returned_session_id}")
        assert final_poll.status_code == 200
        final_data = final_poll.json()
        assert final_data["status"] == "approved"
        assert final_data["response_text"] == approved_text

        # ── Step 6: Queue count decreases ─────────────────────────────────────
        updated_queue = client.get("/hitl/queue")
        # The item was approved; pending count should be 0 (or less than before)
        assert updated_queue.json()["pending_count"] == 0
