def _build_records(pairs: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """Flatten conflict pairs into individual registry records."""
    records = []
    for pair in pairs:
        for rec in pair["records"]:
            records.append(
                {
                    "medication_id": str(uuid.UUID(int=_RNG.getrandbits(128))),
                    "name_generic": pair["name_generic"],
                    "name_brand": pair["name_brand"],
                    "jurisdiction": rec["jurisdiction"],
                    "status": rec["status"],
                    "restriction_reason": rec["restriction_reason"],
                    "therapeutic_alternatives": rec["therapeutic_alternatives"],
                    "last_updated": _rdate(0, 500),
                    "source_registry": rec["source_registry"],
                }
            )
    return records


def generate_medication_registry() -> List[Dict[str, Any]]:
    """
    Return a list of 50 synthetic cross-border medication registry records.

    Records are deterministic (seeded RNG) so repeated calls produce
    identical data with identical UUIDs — good for reproducibility.
    """
    records = _build_records(_CONFLICT_PAIRS)
    # We have 25 pairs × 2 jurisdictions = 50 records exactly.
    assert len(records) == 50, f"Expected 50 records, got {len(records)}"
    return records

def _synthetic_patient_id(index: int) -> str:
    """Generate a deterministic, opaque synthetic hash — not a real ID."""
    raw = f"SYNTH-PATIENT-{index:05d}-DELABYRIX"
    return "P-" + hashlib.sha1(raw.encode()).hexdigest()[:12].upper()


def _weighted_choice(population: list, weights: list, rng: random.Random) -> Any:
    """Single weighted random choice using the provided RNG."""
    return rng.choices(population, weights=weights, k=1)[0]


def _make_record(index: int) -> Dict[str, Any]:
    """Generate a single patient log record."""
    n_conditions = _RNG.randint(1, 3)
    condition_codes = _RNG.sample(_ICD10_CATEGORIES, n_conditions)

    n_meds = _RNG.randint(0, 5)
    medications = _RNG.sample(_MEDICATIONS, n_meds)

    n_diet = _RNG.randint(0, 3)
    diet_flags = _RNG.sample(_DIET_FLAGS, n_diet)

    return {
        "patient_id": _synthetic_patient_id(index),
        "age_band": _weighted_choice(_AGE_BANDS, _AGE_BAND_WEIGHTS, _RNG),
        "nationality": _weighted_choice(_NATIONALITIES, _NATIONALITY_WEIGHTS, _RNG),
        "condition_codes": condition_codes,
        "medications_current": medications,
        "diet_flags": diet_flags,
        "interaction_language": _weighted_choice(_LANG_TAGS, _LANG_WEIGHTS, _RNG),
        "visit_type": _weighted_choice(_VISIT_TYPES, _VISIT_WEIGHTS, _RNG),
    }


def generate_patient_logs(count: int = 200) -> List[Dict[str, Any]]:
    """
    Return a list of ``count`` synthetic patient log records.

    Records are deterministic (seeded RNG) so repeated calls with the same
    ``count`` produce identical data — suitable for reproducible tests.
    No PII is present in any record.
    """
    return [_make_record(i) for i in range(count)]

def _load_from_file(path: Path) -> List[Dict[str, Any]]:
    """Read and parse the registry JSON file. Raises if file is missing or malformed."""
    if not path.exists():

    target = path or _DEFAULT_REGISTRY_PATH

        len(_registry_cache),

def _normalise(text: str) -> str:
    """Lowercase, strip, collapse whitespace."""
    return re.sub(r"\s+", " ", text.strip().lower())


def check_medication(
    medication_name: str,
    jurisdiction: str,
) -> Optional[RegistryCheckResponse]:
    """
    Look up a medication by INN or brand name within a specific jurisdiction.

    Parameters
    ──────────
    medication_name : INN or brand name (case-insensitive, partial match allowed)
    jurisdiction    : "US_FDA" | "EU_EMA"

    Returns
    ───────
    RegistryCheckResponse for the first matching record, or None if not found.
    """
    query = _normalise(medication_name)
    j = jurisdiction.upper()

    for raw in get_registry():
        if raw.get("jurisdiction", "").upper() != j:
            continue
        if query in _normalise(raw.get("name_generic", "")) or query in _normalise(
            raw.get("name_brand", "")
        ):
            record = MedicationRecord.model_validate(raw)
            return RegistryCheckResponse(
                medication_id=record.medication_id,
                name_generic=record.name_generic,
                name_brand=record.name_brand,
                jurisdiction=record.jurisdiction,
                status=record.status,
                restriction_reason=record.restriction_reason,
                therapeutic_alternatives=record.therapeutic_alternatives,
                last_updated=record.last_updated,
                source_registry=record.source_registry,
            )
    return None


def list_medications(
    page: int = 1,
    page_size: int = 20,
    jurisdiction: Optional[str] = None,
) -> RegistryListResponse:
    """
    Return a paginated list of medication registry records.

    Parameters
    ──────────
    page       : 1-based page number
    page_size  : records per page (max 100)
    jurisdiction : optional filter — "US_FDA" | "EU_EMA"

    Returns
    ───────
    RegistryListResponse with total, page, page_size, and records list.
    """
    page_size = min(page_size, 100)
    page = max(page, 1)

    all_records = get_registry()

    if jurisdiction:
        j = jurisdiction.upper()
        all_records = [r for r in all_records if r.get("jurisdiction", "").upper() == j]

    total = len(all_records)
    start = (page - 1) * page_size
    end = start + page_size
    page_records = [MedicationRecord.model_validate(r) for r in all_records[start:end]]

    return RegistryListResponse(
        total=total,
        page=page,
        page_size=page_size,
        records=page_records,
    )
